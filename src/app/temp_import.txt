  // Import impressions and CTR from CSV
  const importCSV = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file) return;
    
    setImportingCSV(true);
    console.log('=== Starting CSV import ===');
    
    try {
      const text = await file.text();
      const lines = text.split('\n');
      
      console.log(`CSV has ${lines.length} total lines`);
      console.log('First 3 lines:', lines.slice(0, 3));
      
      // Skip header row and total row
      const dataLines = lines.slice(2).filter(line => line.trim());
      console.log(`Processing ${dataLines.length} data lines`);
      
      let successCount = 0;
      let notFoundCount = 0;
      let errorCount = 0;
      const notFoundVideos: string[] = [];
      const errors: string[] = [];
      
      for (let i = 0; i < dataLines.length; i++) {
        const line = dataLines[i];
        try {
          // Parse CSV line (handles quoted fields)
          const matches = line.match(/(\".*?\"|[^,]+)(?=\s*,|\s*$)/g);
          
          if (!matches || matches.length < 6) {
            console.log(`Line ${i+3} skipped - not enough columns:`, matches?.length || 0);
            continue;
          }
          
          const videoId = matches[0].replace(/\"/g, '').trim();
          const impressionsStr = matches[4].replace(/\"/g, '').replace(/,/g, '').trim();
          const ctrStr = matches[5].replace(/\"/g, '').replace(/%/g, '').trim();
          const impressions = parseInt(impressionsStr);
          const ctr = parseFloat(ctrStr);
          
          console.log(`Line ${i+3}: videoId="${videoId}", impressions="${impressionsStr}"(${impressions}), ctr="${ctrStr}"(${ctr})`);
          
          // Skip if no valid data
          if (!videoId || isNaN(impressions) || isNaN(ctr)) {
            console.log(`Line ${i+3} skipped - invalid data`);
            continue;
          }
          
          // Check if video exists in our database
          const video = videos.find(v => v.videoId === videoId);
          
          if (!video) {
            notFoundCount++;
            notFoundVideos.push(videoId);
            console.log(`Video not found in database: ${videoId}`);
            continue;
          }
          
          console.log(`→ Updating video ${videoId} with impressions=${impressions}, ctr=${ctr}`);
          
          // Update video with impressions and CTR
          const response = await fetch(
            `https://${projectId}.supabase.co/functions/v1/make-server-6ab9c767/videos/${videoId}/advanced-snapshot`,
            {
              method: "POST",
              headers: {
                "Content-Type": "application/json",
                Authorization: `Bearer ${publicAnonKey}`,
              },
              body: JSON.stringify({
                impressions,
                ctr,
              }),
            }
          );
          
          if (response.ok) {
            const result = await response.json();
            console.log(`✓ Successfully updated ${videoId}`);
            successCount++;
          } else {
            const errorText = await response.text();
            console.error(`✗ Failed to update ${videoId}:`, response.status, errorText);
            errors.push(`${videoId}: ${response.status}`);
            errorCount++;
          }
          
          // Small delay to avoid overwhelming the server
          await new Promise(resolve => setTimeout(resolve, 50));
        } catch (error) {
          console.error(`Error processing line ${i+3}:`, error);
          errors.push(`Line ${i+3}: ${String(error)}`);
          errorCount++;
        }
      }
      
      console.log('=== Import complete ===');
      console.log('Results:', { successCount, notFoundCount, errorCount });
      console.log('Not found videos:', notFoundVideos);
      console.log('Errors:', errors);
      
      // Show results
      let message = `Successfully imported data for ${successCount} video(s)`;
      if (notFoundCount > 0) {
        message += `. ${notFoundCount} video(s) not found in database`;
      }
      if (errorCount > 0) {
        message += `. ${errorCount} error(s)`;
      }
      
      if (successCount > 0) {
        toast.success(message);
        await fetchDatabaseVideos();
      } else {
        toast.error('No data was imported. Check console logs for details.');
      }
      
    } catch (error) {
      console.error('Error importing CSV:', error);
      toast.error(`Failed to import CSV file: ${String(error)}`);
    } finally {
      setImportingCSV(false);
      // Reset file input
      event.target.value = '';
    }
  };
